cmake_minimum_required(VERSION 3.16)
project(savior_perception_cpp LANGUAGES CXX)

# ---------------------------------------------------------------- options
option(SAVIOR_WITH_TRT "TensorRT YOLO26-seg segmenter (needs NvInfer.h + CUDA)" ON)
option(SAVIOR_BUILD_ROS "ROS 2 component + executable (ament)" ON)
option(SAVIOR_BUILD_PYTHON "pybind11 module savior_cpp (offline equivalence harness)" OFF)
set(SAVIOR_OPENCV_DIR "" CACHE PATH "OpenCVConfig.cmake dir (empty = system / JetPack OpenCV)")
set(TENSORRT_ROOT "/opt/tensorrt" CACHE PATH "TensorRT root (include/, lib/); JetPack: headers/libs in the system dirs")

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)  # OpenCV / TensorRT outside the default linker path
# no FMA contraction in the C++ core: same results on x86_64 and aarch64 (and closest to numpy's scalar arithmetic)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-ffp-contract=off> $<$<COMPILE_LANGUAGE:CXX>:-Wall>)

if(SAVIOR_OPENCV_DIR)
  set(OpenCV_DIR ${SAVIOR_OPENCV_DIR})
  find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d video features2d imgcodecs NO_DEFAULT_PATH PATHS ${SAVIOR_OPENCV_DIR})
else()
  find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d video features2d imgcodecs)
endif()
message(STATUS "savior_perception_cpp: OpenCV ${OpenCV_VERSION} (${OpenCV_DIR})")
find_package(yaml-cpp REQUIRED)
if(TARGET yaml-cpp::yaml-cpp)
  set(YAMLCPP_TARGET yaml-cpp::yaml-cpp)
else()
  set(YAMLCPP_TARGET ${YAML_CPP_LIBRARIES})
endif()

# ---------------------------------------------------------------- TensorRT / CUDA
if(SAVIOR_WITH_TRT)
  find_path(TRT_INCLUDE NvInfer.h HINTS ${TENSORRT_ROOT}/include /usr/include/${CMAKE_LIBRARY_ARCHITECTURE} /usr/include)
  find_library(TRT_NVINFER nvinfer HINTS ${TENSORRT_ROOT}/lib /usr/lib/${CMAKE_LIBRARY_ARCHITECTURE})
  if(NOT TRT_INCLUDE OR NOT TRT_NVINFER)
    message(WARNING "TensorRT not found: building without the YOLO segmenter (SAVIOR_WITH_TRT=OFF)")
    set(SAVIOR_WITH_TRT OFF)
  else()
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
      if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
        set(CMAKE_CUDA_ARCHITECTURES 87)          # Jetson Orin (NX / AGX / Nano)
      else()
        set(CMAKE_CUDA_ARCHITECTURES "86;89;120")  # Ampere / Ada / Blackwell desktop
      endif()
    endif()
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    message(STATUS "savior_perception_cpp: TensorRT ${TRT_INCLUDE} ${TRT_NVINFER}, CUDA archs ${CMAKE_CUDA_ARCHITECTURES}")
  endif()
endif()

# ---------------------------------------------------------------- core library (no ROS)
add_library(savior_core SHARED
  src/types.cpp src/config.cpp src/roles.cpp src/calibration.cpp src/stereo_depth.cpp src/gating.cpp src/motion.cpp
  src/selective.cpp src/tracker.cpp src/extension.cpp src/compose.cpp src/fillpoly411.cpp src/pipeline.cpp)
target_include_directories(savior_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(savior_core PUBLIC ${OpenCV_LIBS} ${YAMLCPP_TARGET})
target_include_directories(savior_core PUBLIC ${OpenCV_INCLUDE_DIRS})
if(SAVIOR_WITH_TRT)
  add_library(savior_yolo SHARED src/yolo_trt.cpp cuda/yolo_kernels.cu)
  target_include_directories(savior_yolo PUBLIC ${TRT_INCLUDE})
  target_link_libraries(savior_yolo PUBLIC savior_core ${TRT_NVINFER} CUDA::cudart)
  target_compile_definitions(savior_yolo PUBLIC SAVIOR_WITH_TRT)
  target_compile_options(savior_yolo PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wno-deprecated-declarations>)
  set_target_properties(savior_yolo PROPERTIES CUDA_SEPARABLE_COMPILATION OFF)
endif()

# ---------------------------------------------------------------- python module (harness)
if(SAVIOR_BUILD_PYTHON)
  find_package(Python3 COMPONENTS Interpreter Development REQUIRED)
  execute_process(COMMAND ${Python3_EXECUTABLE} -c "import pybind11; print(pybind11.get_cmake_dir())"
                  OUTPUT_VARIABLE PYBIND11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
  find_package(pybind11 CONFIG REQUIRED HINTS ${PYBIND11_DIR})
  pybind11_add_module(savior_cpp python/bindings.cpp)
  target_link_libraries(savior_cpp PRIVATE savior_core)
  if(SAVIOR_WITH_TRT)
    target_link_libraries(savior_cpp PRIVATE savior_yolo)
  endif()
  set_target_properties(savior_cpp PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()

# ---------------------------------------------------------------- ROS 2 component + executable
if(SAVIOR_BUILD_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(rclcpp_components REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(message_filters REQUIRED)
  find_package(ament_index_cpp REQUIRED)
  add_library(savior_node_component SHARED src/savior_node.cpp)
  target_link_libraries(savior_node_component savior_core)
  if(SAVIOR_WITH_TRT)
    target_link_libraries(savior_node_component savior_yolo)
  endif()
  ament_target_dependencies(savior_node_component rclcpp rclcpp_components sensor_msgs std_msgs message_filters ament_index_cpp)
  rclcpp_components_register_node(savior_node_component PLUGIN "savior_perception_cpp::SaviorNode" EXECUTABLE savior_node_cpp)
  set(_targets savior_core savior_node_component)
  if(SAVIOR_WITH_TRT)
    list(APPEND _targets savior_yolo)
  endif()
  install(TARGETS ${_targets} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
  install(DIRECTORY include/ DESTINATION include)
  install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
  if(SAVIOR_BUILD_PYTHON)
    install(TARGETS savior_cpp LIBRARY DESTINATION lib/python3/dist-packages)
  endif()
  ament_package()
else()
  install(TARGETS savior_core LIBRARY DESTINATION lib)
  if(SAVIOR_WITH_TRT)
    install(TARGETS savior_yolo LIBRARY DESTINATION lib)
  endif()
  if(SAVIOR_BUILD_PYTHON)
    install(TARGETS savior_cpp LIBRARY DESTINATION lib)
  endif()
endif()
